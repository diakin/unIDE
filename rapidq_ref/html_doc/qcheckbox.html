 QCheckbox Examples 
  DIM Form AS QForm
  DIM Checkbox as QCheckBox

  SUB Check
    IF CheckBox.Checked THEN
       SHOWMESSAGE "Checked!"
    END IF
  END SUB

  CheckBox.Parent = Form
  CheckBox.OnClick = Check

  Form.ShowModal
