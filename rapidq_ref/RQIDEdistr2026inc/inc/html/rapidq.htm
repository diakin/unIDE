'-- ***********************************************************************' 
sub AddClrString (AddString$, Clr, Sender as QRICHEDIT)
'Add colored string to QRICHEDIT 
          Sender.SelStart =len (Sender.Text)
          Sender.SelLength =Len(AddString$)
          Sender.SelAttributes.Color=Clr
          Sender.AddStrings AddString$
end sub
