QLine Example

$TypeCheck on
$Include "Object\QLine.inc"

CREATE Form AS QFORM
    Caption = "Line"
    Width = 320
    Height = 240
    Center
    CREATE Menu as QMainMenu
      CREATE Menu1 as QMenuItem
        caption="Fichier"
      END CREATE
    END CREATE
    Create Line as QLine
      Align=1
      Left = 10
      Top = 10
    End Create
END CREATE

Form.ShowModal
